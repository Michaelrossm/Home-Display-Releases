cmake_minimum_required(VERSION 3.20)
project(home_lua_background LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(LUA "${ROOT}/managed_components/espressif__lua/lua")
add_library(real_lua STATIC "${LUA}/onelua.c")
target_compile_definitions(real_lua PRIVATE MAKE_LIB)
target_include_directories(real_lua PUBLIC "${LUA}")
add_executable(background_check check.cpp "${ROOT}/src/apps/lua/LuaBackgroundRun.cpp" "${ROOT}/src/apps/lua/LuaEvents.cpp")
target_include_directories(background_check PRIVATE "${ROOT}/src/apps/lua" "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(background_check PRIVATE real_lua)
if(NOT WIN32)
  target_link_libraries(real_lua PUBLIC m)
endif()
