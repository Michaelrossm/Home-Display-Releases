cmake_minimum_required(VERSION 3.20)
project(home_image_codec_checks C CXX)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(WEBP "${ROOT}/components/home_image_codecs/libwebp")
set(PNG "${ROOT}/managed_components/espressif__libpng")
set(ZLIB "${ROOT}/managed_components/espressif__zlib/zlib")
file(GLOB WEBP_SOURCES "${WEBP}/src/dec/*.c" "${WEBP}/src/dsp/*.c" "${WEBP}/src/utils/*.c")
file(GLOB PNG_SOURCES "${PNG}/libpng/*.c" "${PNG}/libpng/intel/*.c")
list(FILTER PNG_SOURCES EXCLUDE REGEX "/(pngtest|example|pngvalid)\\.c$")
file(GLOB ZLIB_SOURCES "${ZLIB}/*.c")
add_library(host_webp STATIC ${WEBP_SOURCES})
target_include_directories(host_webp PRIVATE "${WEBP}")
target_compile_definitions(host_webp PRIVATE malloc=hp_codec_malloc calloc=hp_codec_calloc realloc=hp_codec_realloc free=hp_codec_free _CRT_SECURE_NO_WARNINGS)
add_library(host_png STATIC ${PNG_SOURCES} ${ZLIB_SOURCES})
# A fresh Component Manager download has no generated pnglibconf.h. Keep the
# host configuration in the build tree, never modify managed dependencies.
configure_file("${PNG}/libpng/scripts/pnglibconf.h.prebuilt"
               "${CMAKE_CURRENT_BINARY_DIR}/pnglibconf.h" COPYONLY)
target_include_directories(host_png PUBLIC "${CMAKE_CURRENT_BINARY_DIR}" "${PNG}" "${PNG}/libpng" "${ZLIB}")
target_compile_definitions(host_png PRIVATE _CRT_SECURE_NO_WARNINGS)
add_executable(codec_check check.cpp "${ROOT}/src/services/ImageCodec.cpp")
target_compile_features(codec_check PRIVATE cxx_std_17)
target_include_directories(codec_check PRIVATE "." "${ROOT}/components/home_image_codecs" "${WEBP}/src")
target_link_libraries(codec_check PRIVATE host_webp host_png)
if(UNIX)
  target_link_libraries(codec_check PRIVATE m)
endif()
